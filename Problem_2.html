Average maximum temperature by month and year

plot_ly(
  data = monthly_temp,
  x = ~month,
  y = ~year,
  z = ~avg_tmax,
  type = "heatmap",
  colorbar = list(title = "°C")
) |>
  layout(
    xaxis = list(title = "Month", tickmode = "linear", dtick = 1),
    yaxis = list(title = "Year", tickmode = "linear", dtick = 1)
  )

Maximum temperature by season

plot_ly(
  data = weather_df |> filter(!is.na(tmax)),
  x = ~season,
  y = ~tmax,
  color = ~season,
  type = "box",
  boxpoints = FALSE
) |>
  layout(
    showlegend = FALSE,
    xaxis = list(title = "Season"),
    yaxis = list(title = "Maximum temperature (°C)")
  )

Daily maximum vs. minimum temperature

plot_ly(
  data = weather_df |>
    filter(!is.na(tmax), !is.na(tmin)) |>
    slice_sample(n = 500),
  x = ~tmin,
  y = ~tmax,
  color = ~season,
  type = "scatter",
  mode = "markers",
  marker = list(opacity = 0.6)
) |>
  layout(
    xaxis = list(title = "Minimum temperature (°C)"),
    yaxis = list(title = "Maximum temperature (°C)")
  )
